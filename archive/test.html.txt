<body>
  <h1>Test Währungsrechner mit PostMessage</h1>

  <div class="controls">
    <label for="from">Von:</label>
    <select id="from"></select>

    <label for="to">Nach:</label>
    <select id="to"></select>

    <label for="period">Historie:</label>
    <select id="period">
      <option value="7">Letzte 7 Tage</option>
      <option value="30" selected>Letzte 30 Tage</option>
    </select>

    <button id="send">Anfrage senden</button>
  </div>

  <iframe id="conv" src="chart.html"></iframe>

  <h2>📝 Log</h2>
  <div id="log">Warte auf Aktionen…</div>

  <script>
    const iframe = document.getElementById("conv");
    const log    = document.getElementById("log");
    const selFrom   = document.getElementById("from");
    const selTo     = document.getElementById("to");
    const selPeriod = document.getElementById("period");

    // Initial: Basis- und Gegenwährungen füllen, indem wir einmal Live-Raten vom iframe holen
    window.addEventListener("message", function initListener(event) {
      const { type, payload } = event.data || {};
      if (type === "fxRatesUpdated") {
        // Währungen aus payload.rates nutzen
        const codes = Object.keys(payload.rates).sort();
        [selFrom, selTo].forEach(sel => {
          sel.innerHTML = "";
          codes.forEach(c => {
            const opt = document.createElement("option");
            opt.value = c; opt.textContent = c;
            sel.appendChild(opt);
          });
        });
        selFrom.value = selFrom.querySelector("option").value;
        selTo.value   = selTo.querySelectorAll("option")[1]?.value || selFrom.value;
        window.removeEventListener("message", initListener);
        log.textContent += "[Init] Währungen geladen aus fxRatesUpdated\n";
      }
    });

    // Bei Klick: senden wir die Anfrage mit from/to/period
    document.getElementById("send").addEventListener("click", () => {
      const from   = selFrom.value;
      const to     = selTo.value;
      const period = parseInt(selPeriod.value, 10);
      log.textContent += `[Anfrage] getFxRates from=${from}, to=${to}, period=${period} an ${new Date().toLocaleTimeString()}\n`;
      iframe.contentWindow.postMessage({
        type: "getFxRates",
        payload: { from, to, period }
      }, "*");
    });

    // Antworten protokollieren (rates + history)
    window.addEventListener("message", (event) => {
      const { type, payload } = event.data || {};
      if (type === "fxData") {
        const time = new Date(payload.timestamp).toLocaleTimeString();
        log.textContent += `[Antwort] fxData um ${time}\n`;
        log.textContent += `  Rates: ${JSON.stringify(payload.rates)}\n`;
        log.textContent += `  History (${payload.history.length} Kerzen):\n`;
        payload.history.forEach(c => {
          const d = new Date(c.x).toLocaleDateString();
          const [o,h,l,cl] = c.y.map(v => v.toFixed(4));
          log.textContent += `    ${d} O=${o}, H=${h}, L=${l}, C=${cl}\n`;
        });
        log.scrollTop = log.scrollHeight;
      }
    });

    // Starte mit einer Standard-Anfrage, damit initListener feuert
    iframe.addEventListener("load", () => {
      iframe.contentWindow.postMessage({ type: "getFxRates", payload: {} }, "*");
    });
  </script>
